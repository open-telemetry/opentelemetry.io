Each call renders where it stands; a page's calls must name the same data. Fails the build on a positional or unknown argument, a data_url that is neither an https URL nor a set page param, or a second data source. Guide: https://opentelemetry.io/site/implementation/plugins-and-scripts/#benchmarks @param {string} data_url The data.js URL, https, or param:NAME naming a page param that holds it. @param {string} [filter] Comma-separated strings; a bench is charted when its extra contains any of them. @param {string} [group_separator] Splits extra into a group, joined to the bench name as the chart's title, and a series name; an extra without it is both. Absent, each bench is its own chart. @param {string} [series_separator] Trims the series name at its first occurrence; nothing without group_separator. @example {{< benchmarks data_url="param:otel-benchmarks.data_url" filter="10kDPS,10kSPS" group_separator="/" series_separator=" - " >}} */ -}} {{ $docs := "https://opentelemetry.io/site/implementation/plugins-and-scripts/#benchmarks" -}} {{ if or (not .Params) (not .IsNamedParams) -}} {{ errorf "%s: shortcode %q: takes named arguments only; see %s" .Position .Name $docs -}} {{ else -}} {{ $src := .Get "data_url" | default "" -}} {{ $data := $src -}} {{ if hasPrefix $src "param:" -}} {{ $data = .Page.Param (strings.TrimPrefix "param:" $src) | default "" -}} {{ end -}} {{ $url := urls.Parse $data -}} {{ $prior := .Page.Store.Get "otelBenchmarks" -}} {{ if and (ne $data $src) (not $data) -}} {{ errorf "%s: shortcode %q: page param %q, named by `data_url`, is not set; see %s" .Position .Name (strings.TrimPrefix "param:" $src) $docs -}} {{ else if or (ne $url.Scheme "https") (not $url.Hostname) -}} {{ errorf "%s: shortcode %q: `data_url` must be an https URL or `param:NAME` naming one, got %q; see %s" .Position .Name $data $docs -}} {{ else if and $prior (ne $prior $data) -}} {{ errorf "%s: shortcode %q: `data_url` differs from an earlier call's (%s): one data source per page; see %s" .Position .Name $prior $docs -}} {{ else -}} {{ .Page.Store.Set "otelBenchmarks" $data -}} {{ end -}} {{ $options := dict -}} {{ range $name, $value := .Params -}} {{ if eq $name "filter" -}} {{ with complement (slice "") (apply (split $value ",") "strings.TrimSpace" ".") -}} {{ $options = merge $options (dict $name .) -}} {{ end -}} {{ else if in (slice "group_separator" "series_separator") $name -}} {{ $options = merge $options (dict $name $value) -}} {{ else if ne $name "data_url" -}} {{ errorf "%s: shortcode %q: unknown argument %q; see %s" $.Position $.Name $name $docs -}} {{ end -}} {{ end -}}
